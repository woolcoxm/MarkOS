# MarkOS boot config for real Raspberry Pi hardware (Pi 4 / Pi 5).
# Lives in the boot partition next to kernel8.img and the stock firmware
# blobs (start4.elf, fixup4.dat, etc. — copied from the official
# raspberry-pi/firmware repository, which `make sd` automates later).

arm_64bit=1
# Force the firmware's 64-bit load address to match the linker script.
kernel_address=0x80000
# Debug console on GPIO 14/15 (PL011), 115200 8N1.
enable_uart=1
