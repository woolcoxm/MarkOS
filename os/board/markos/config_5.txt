# MarkOS boot config for Raspberry Pi 5 (BCM2712).
# Lives on the boot (FAT32) partition next to the firmware.

kernel=Image
disable_overscan=1

# Full 2.4 GHz on all four A76 cores — inference is sustained CPU load.
arm_boost=1

# Debug/recovery console on the J8 UART (GPIO14/15, 115200 8N1).
enable_uart=1
dtparam=uart0=on

# Hardware watchdog device (/dev/watchdog0) for the last-resort reboot.
dtparam=watchdog=on

# PCIe for NVMe HATs. Gen 3 is stable on most drives; uncomment on flaky
# link trains (see docs/recovery.md).
# dtparam=pciex1_gen=3

# Boot aggressively but deterministically: no silent display, fast boot.
boot_delay=0
