# MarkOS boot config for Raspberry Pi 5 (BCM2712).
# Lives on the boot (FAT32) partition next to the firmware. Style mirrors
# the official Raspberry Pi OS config — the boot environment is vendored
# from there (see post-image.sh) and known-good on d0-stepping boards.

kernel=Image
arm_64bit=1
disable_overscan=1
disable_fw_kms_setup=1

# Full 2.4 GHz on all four A76 cores — inference is sustained CPU load.
arm_boost=1

# Debug/recovery console on the J8 UART (GPIO14/15, 115200 8N1).
enable_uart=1
dtparam=uart0=on

# Hardware watchdog device (/dev/watchdog0) for the last-resort reboot.
dtparam=watchdog=on

# PCIe: required for NVMe HATs AND the Axera AX8850 M.2 card (AXCL setup
# docs; without pciex1 + the compat overlay the EP is not recognized /
# MSI IRQs cannot be allocated — axcl-samples#5). Gen 3 for NVMe drives.
dtparam=pciex1
dtparam=pciex1_gen=3
dtoverlay=pciex1-compat-pi5,no-mip
